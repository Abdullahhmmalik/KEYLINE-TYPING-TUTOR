cmake_minimum_required(VERSION 3.18)
project(Keyline VERSION 1.0.2 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(KEYLINE_BUILD_GUI "Build the raylib desktop application" ON)
option(KEYLINE_FETCH_RAYLIB "Download pinned raylib if not installed" ON)
set(RAYLIB_SOURCE_DIR "" CACHE PATH "Optional existing raylib 5.5 source directory")

add_library(keyline_core STATIC src/config.c src/stats.c src/typing.c src/data.c src/achievements.c)
target_include_directories(keyline_core PUBLIC src)
if(NOT WIN32)
    target_link_libraries(keyline_core PUBLIC m)
endif()
function(keyline_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wstrict-prototypes)
    endif()
endfunction()
keyline_warnings(keyline_core)

include(CTest)
if(BUILD_TESTING)
    add_executable(keyline_tests tests/test_core.c)
    target_link_libraries(keyline_tests PRIVATE keyline_core)
    keyline_warnings(keyline_tests)
    add_test(NAME core COMMAND keyline_tests "${CMAKE_CURRENT_BINARY_DIR}/test-data")
    # Use a unique directory for every CTest run, so repeated runs stay independent.
    set_tests_properties(core PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
endif()

if(KEYLINE_BUILD_GUI)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
    if(RAYLIB_SOURCE_DIR)
        add_subdirectory("${RAYLIB_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/raylib")
    elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/raylib-5.5/CMakeLists.txt")
        add_subdirectory(vendor/raylib-5.5 "${CMAKE_CURRENT_BINARY_DIR}/raylib")
    elseif(MINGW AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/raylib-5.5_win64_mingw-w64/lib/libraylib.a")
        add_library(raylib STATIC IMPORTED)
        set_target_properties(raylib PROPERTIES
            IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/vendor/raylib-5.5_win64_mingw-w64/lib/libraylib.a"
            INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/raylib-5.5_win64_mingw-w64/include"
            INTERFACE_LINK_LIBRARIES "opengl32;gdi32;winmm")
    else()
        find_package(raylib 5.5 QUIET)
        if(NOT raylib_FOUND)
            if(NOT KEYLINE_FETCH_RAYLIB)
                message(FATAL_ERROR "Install raylib 5.5 or enable KEYLINE_FETCH_RAYLIB.")
            endif()
            include(FetchContent)
            FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.zip)
            FetchContent_MakeAvailable(raylib)
        endif()
    endif()
    add_executable(keyline WIN32 src/main.c src/app.c src/ui.c src/desktop.c)
    target_compile_definitions(keyline PRIVATE APP_BUILD_TAG="cmake-${PROJECT_VERSION}")
    target_link_libraries(keyline PRIVATE keyline_core raylib)
    keyline_warnings(keyline)
    if(MINGW)
        target_link_options(keyline PRIVATE -static -static-libgcc)
    endif()
    install(TARGETS keyline RUNTIME DESTINATION .)
endif()
